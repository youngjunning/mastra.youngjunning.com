const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "inceptron/zai-org/GLM-5.3-Flash"
      : "inceptron/deepseek-ai/DeepSeek-V4-Flash-0731";
  }
});
