const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "ai-router/gpt-5.6-terra"
      : "ai-router/gpt-5.4";
  }
});
