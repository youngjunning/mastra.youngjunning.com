const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "arcee/zai-org/glm-5.3-flash"
      : "arcee/deepseek/deepseek-v4-flash-latest";
  }
});
