```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "custom-agent",
  name: "custom-agent",
  model: {
    url: "https://inference.generativeai.us-chicago-1.oci.oraclecloud.com/openai/v1",
    id: "oci/meta.llama-3.3-70b-instruct",
    apiKey: process.env.OCI_GENAI_API_KEY,
    headers: {
      "X-Custom-Header": "value"
    }
  }
});
```

### 動態選擇模型 {/* #dynamic-model-selection */}

```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "oci/xai.grok-4.6"
      : "oci/meta.llama-3.3-70b-instruct";
  }
});
```
