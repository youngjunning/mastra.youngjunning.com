import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "custom-agent",
  name: "custom-agent",
  model: {
    url: "https://api.pendra.ai/api/v1",
    id: "pendra/deepseek-v4-flash",
    apiKey: process.env.PENDRA_API_KEY,
    headers: {

});

### 動態選擇模型 {/* #dynamic-model-selection */}

```typescript title="src/mastra/agents/my-agent.ts"
import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "pendra/qwen3.6:27b"
      : "pendra/deepseek-v4-flash";
  }
});
```
