import { Agent } from "@mastra/core/agent";

const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "kimi-code-plan-global/kimi-for-coding-highspeed"
      : "kimi-code-plan-global/k3";
  }
});
