const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "submodel/zai-org/GLM-4.5-FP8"
      : "submodel/Qwen/Qwen3-235B-A22B-Instruct-2507";
  }
});
