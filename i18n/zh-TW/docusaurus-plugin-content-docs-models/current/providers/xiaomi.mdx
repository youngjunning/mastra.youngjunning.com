const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "xiaomi/mimo-v2.6-pro-ultraspeed"
      : "xiaomi/mimo-v2.5";
  }
});
