const agent = new Agent({
  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "runinfra/zai-org/GLM-5.3-Flash"
      : "runinfra/Inferact/Qwen3.8-2.4T-A95B-NVFP4";
  }
});
