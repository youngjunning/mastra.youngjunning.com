  id: "dynamic-agent",
  name: "Dynamic Agent",
  model: ({ requestContext }) => {
    const useAdvanced = requestContext.task === "complex";
    return useAdvanced
      ? "munito-eu/zai-org/glm-5.3-flash"
      : "munito-eu/aleph-alpha/kolibri-1";
  }
});
